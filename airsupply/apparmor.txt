#include <tunables/global>

profile airsupply flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Minimum for Bluetooth adapter use.
  capability net_admin,
  capability net_raw,

  # s6-overlay and shell. `rix` on /init because it is a shell script that
  # /bin/sh must read to interpret.
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/lib/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /run/** rwk,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /etc/cont-init.d/** rwix,

  # D-Bus socket -- the only route to BlueZ.
  /run/dbus/system_bus_socket rw,

  # BlueZ calls back into our exported Profile1 object using our dynamic unique
  # bus name (:1.xxx), which cannot be allowlisted by peer name. Broad
  # send/receive on the system bus is the standard pattern for add-ons that
  # talk to BlueZ.
  dbus send    bus=system,
  dbus receive bus=system,

  # Persistent state: the SRP masterPairKey, once pairing works.
  /data/ r,
  /data/** rwk,
  /tmp/** rwk,

  # Python runtime and application code. /usr/src/ needs its own rule -- /**
  # covers children but not the directory, and import must scandir the parent.
  /usr/lib/python3*/** r,
  /usr/bin/python3* ix,
  /usr/src/ r,
  /usr/src/** r,

  /lib/** r,
  /usr/lib/libbluetooth* r,
  /usr/lib/libdbus* r,
  /usr/share/** r,
  /etc/resolv.conf r,
  /etc/nsswitch.conf r,
  /etc/hosts r,
  /etc/host.conf r,

  /proc/*/status r,
  /sys/class/bluetooth/** r,
  /sys/devices/**/bluetooth/** r,
  /sys/devices/**/idVendor r,
  /sys/devices/**/idProduct r,
  /sys/devices/**/manufacturer r,
  /sys/devices/**/product r,

  # `network bluetooth` is the rule that makes AF_BLUETOOTH sockets possible at
  # all. It is kept even though the intended transport is a file descriptor
  # handed over by BlueZ, because that fd is a Bluetooth socket and operations
  # on it are mediated by this same rule.
  network inet stream,
  network inet dgram,
  network unix stream,
  network unix dgram,
  network bluetooth,

  # Never touch raw HCI. All Bluetooth work goes through BlueZ over D-Bus, so
  # that Home Assistant's passive BLE scanning is not disturbed.
  deny /dev/hci* rw,
  deny /sys/kernel/** rw,
  deny /proc/sys/** w,
}
